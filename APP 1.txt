export default function App() {
  const [lang, setLang] = useState(null); 
  const [tab, setTab] = useState("home");
  
  // Dil seçildiğinde otomatik o dilin verileri yüklenir
  const t = lang ? translations[lang] : translations.tr;
  
  const [tasks, setTasks] = useState([]);
  const [transactions, setTransactions] = useState([]);
  
  const [modalType, setModalType] = useState(null);
  const [newTitle, setNewTitle] = useState("");
  const [newTime, setNewTime] = useState("");
  const [newAmount, setNewAmount] = useState("");
  const [assistantInput, setAssistantInput] = useState("");

  // Dil ilk seçildiğinde listeleri o dile göre doldur
  const handleSelectLanguage = (code) => {
    setLang(code);
    setTasks(translations[code].defaultTasks);
    setTransactions(translations[code].defaultTransactions);
  };

  if (!lang) {
    return (
      <SafeAreaView style={styles.safeCenter}>
        <StatusBar barStyle="light-content"/>
        <View style={styles.welcomeBox}>
          <Ionicons name="globe-outline" size={48} color="#9B8CFF" style={{alignSelf: "center", marginBottom: 16}} />
          <Text style={styles.welcomeTitle}>Dilinizi Seçin / Choose Language</Text>
          
          <View style={styles.langList}>
            {[
              ["tr", "🇹🇷 Türkçe"],
              ["en", "🇬🇧 English"],
              ["es", "🇪🇸 Español"],
              ["zh", "🇨🇳 中文"],
              ["hi", "🇮🇳 हिन्दी"]
            ].map(([code, label]) => (
              <Pressable key={code} style={styles.langOption} onPress={() => handleSelectLanguage(code)}>
                <Text style={styles.langOptionText}>{label}</Text>
                <Ionicons name="chevron-forward" size={18} color="#9B8CFF" />
              </Pressable>
            ))}
          </View>
        </View>
      </SafeAreaView>
    );
  }

  const completed = tasks.filter(item => item.done).length;
  const progress = tasks.length ? completed / tasks.length : 0;
  const balance = transactions.reduce((sum, item) => sum + Number(item.amount), 0);

  const toggleTask = id =>
    setTasks(ts => ts.map(item => item.id === id ? { ...item, done: !item.done } : item));

  const navItems = [
    ["home", t.home, "home-outline"],
    ["assistant", t.assistant, "sparkles-outline"],
    ["calendar", t.calendar, "calendar-outline"],
    ["wallet", t.wallet, "wallet-outline"],
    ["profile", t.profile, "person-outline"]
  ];

  const Home = () => (
    <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.content}>
      <View style={styles.topRow}>
        <View>
          <Text style={styles.muted}>{t.hello}</Text>
          <Text style={styles.title}>{t.title}</Text>
        </View>
        <Pressable style={styles.iconButton} onPress={() => setLang(null)}>
          <Ionicons name="globe-outline" size={22} color="#F7F8FF" />
        </Pressable>
      </View>

      <View style={styles.hero}>
        <View style={{flex:1}}>
          <Text style={styles.heroKicker}>{t.summary}</Text>
          <Text style={styles.heroTitle}>{completed}/{tasks.length} {t.completed}</Text>
          <Text style={styles.heroText}>{t.subText}</Text>
        </View>
        <View style={styles.progressCircle}>
          <Text style={styles.progressText}>{Math.round(progress*100)}%</Text>
        </View>
      </View>

      <View style={styles.sectionHeader}>
        <Text style={styles.sectionTitle}>{t.tasksTitle}</Text>
        <Pressable onPress={() => { setNewTitle(""); setNewTime(""); setModalType("task"); }}>
          <Text style={styles.linkText}>{t.addTask}</Text>
        </Pressable>
      </View>

      {tasks.map(item => (
        <Pressable key={item.id} onPress={() => toggleTask(item.id)}>
          <View style={[styles.card, styles.taskCard]}>
            <View style={[styles.check, item.done && styles.checkDone]}>
              {item.done && <Ionicons name="checkmark" size={16} color="#fff" />}
            </View>
            <View style={{flex:1}}>
              <Text style={[styles.taskTitle, item.done && styles.done]}>{item.title}</Text>
              <Text style={styles.taskMeta}>{item.time}</Text>
            </View>
          </View>
        </Pressable>
      ))}
    </ScrollView>
  );

  const AssistantScreen = () => (
    <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
      <Text style={styles.eyebrow}>{t.aiTitle}</Text>
      <Text style={styles.title}>{t.aiQuestion}</Text>
      <TextInput
        value={assistantInput}
        onChangeText={setAssistantInput}
        placeholder={t.aiPlaceholder}
        placeholderTextColor="#737A91"
        style={styles.input}
      />
      {assistantInput ? (
        <View style={styles.card}>
          <Text style={styles.taskTitle}>{t.aiDetected}</Text>
          <Text style={styles.taskMeta}>{assistantInput}</Text>
        </View>
      ) : null}
    </ScrollView>
  );

  const Calendar = () => (
    <ScrollView contentContainerStyle={styles.content}>
      <Text style={styles.title}>{t.calendar}</Text>
      <View style={styles.card}>
        <Text style={styles.taskTitle}>Eylül 2026</Text>
        <Text style={styles.taskMeta}>Planlar aktif.</Text>
      </View>
    </ScrollView>
  );

  const Wallet = () => (
    <ScrollView contentContainerStyle={styles.content}>
      <Text style={styles.title}>{t.wallet}</Text>
      <View style={styles.card}>
        <Text style={styles.muted}>{t.totalBalance}</Text>
        <Text style={styles.balanceText}>{balance} TL</Text>
        <View style={styles.walletBtnRow}>
          <Pressable style={styles.miniBtnIncome} onPress={() => { setNewTitle(""); setNewAmount(""); setModalType("income"); }}>
            <Ionicons name="add-circle-outline" size={18} color="#fff" />
            <Text style={styles.miniBtnText}>{t.addIncome}</Text>
          </Pressable>
          <Pressable style={styles.miniBtnExpense} onPress={() => { setNewTitle(""); setNewAmount(""); setModalType("expense"); }}>
            <Ionicons name="remove-circle-outline" size={18} color="#fff" />
            <Text style={styles.miniBtnText}>{t.addExpense}</Text>
          </Pressable>
        </View>
      </View>

      <Text style={styles.sectionTitle}>{t.transactions}</Text>
      {transactions.map(item => (
        <View key={item.id} style={[styles.card, styles.taskCard]}>
          <Ionicons 
            name={item.type === "income" ? "arrow-down-circle-outline" : "arrow-up-circle-outline"} 
            size={22} 
            color={item.type === "income" ? "#4ADE80" : "#FF8C8C"} 
          />
          <View style={{flex:1}}>
            <Text style={styles.taskTitle}>{item.title}</Text>
          </View>
          <Text style={[styles.amountText, { color: item.type === "income" ? "#4ADE80" : "#FF8C8C" }]}>
            {item.type === "income" ? `+${item.amount}` : item.amount} TL
          </Text>
        </View>
      ))}
    </ScrollView>
  );

  const Profile = () => (
    <ScrollView contentContainerStyle={styles.content}>
      <Text style={styles.title}>{t.profile}</Text>
      <View style={styles.card}>
        <Text style={styles.taskTitle}>{t.profileTitle}</Text>
        <Text style={styles.taskMeta}>{t.version}</Text>
        <Pressable style={styles.langChangeBtn} onPress={() => setLang(null)}>
          <Ionicons name="globe-outline" size={16} color="#9B8CFF" />
          <Text style={styles.langChangeText}>{t.changeLang}</Text>
        </Pressable>
      </View>
    </ScrollView>
  );

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="light-content"/>
      <View style={{flex:1}}>
        {tab==="home" ? <Home/> : tab==="assistant" ? <AssistantScreen/> : tab==="calendar" ? <Calendar/> : tab==="wallet" ? <Wallet/> : <Profile/>}
        
        <View style={styles.nav}>
          {navItems.map(([id,label,icon])=><Pressable key={id} style={styles.navItem} onPress={()=>setTab(id)}>
            <Ionicons name={tab===id ? icon.replace("-outline","") : icon} size={22} color={tab===id ? "#9B8CFF" : "#777E94"}/>
            <Text style={[styles.navLabel,tab===id&&styles.navActive]}>{label}</Text>
          </Pressable>)}
        </View>
      </View>

      <Modal visible={modalType !== null} animationType="fade" transparent={true}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalBox}>
            <Text style={styles.modalTitle}>
              {modalType === "task" ? t.addTask : modalType === "income" ? t.addIncome : t.addExpense}
            </Text>
            
            <TextInput
              style={styles.input}
              placeholder={modalType === "task" ? t.taskPlaceholder : modalType === "income" ? t.incomePlaceholder : t.expensePlaceholder}
              placeholderTextColor="#737A91"
              value={newTitle}
              onChangeText={setNewTitle}
            />

            {modalType === "task" ? (
              <TextInput
                style={styles.input}
                placeholder={t.timePlaceholder}
                placeholderTextColor="#737A91"
                value={newTime}
                onChangeText={setNewTime}
              />
            ) : (
              <TextInput
                style={styles.input}
                placeholder={t.amountPlaceholder}
                placeholderTextColor="#737A91"
                keyboardType="numeric"
                value={newAmount}
                onChangeText={setNewAmount}
              />
            )}

            <View style={styles.modalButtons}>
              <Pressable style={[styles.modalBtn, styles.cancelBtn]} onPress={() => setModalType(null)}>
                <Text style={styles.btnText}>{t.cancel}</Text>
              </Pressable>
              <Pressable 
                style={[styles.modalBtn, styles.saveBtn]} 
                onPress={() => {
                  if (!newTitle.trim()) return;
                  if (modalType === "task") {
                    setTasks([{ id: Date.now().toString(), title: newTitle, time: newTime || "Gün içi", done: false }, ...tasks]);
                  } else {
                    const val = Number(newAmount) || 0;
                    setTransactions([{ id: Date.now().toString(), title: newTitle, amount: modalType === "income" ? val : -val, type: modalType }, ...transactions]);
                  }
                  setNewTitle(""); setNewTime(""); setNewAmount(""); setModalType(null);
                }}
              >
                <Text style={styles.btnText}>{t.save}</Text>
              </Pressable>
            </View>
          </View>
        </View>
      </Modal>
    </SafeAreaView>
  );
}
